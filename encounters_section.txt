X.toggleRoster=function(field,id,checked){
  const s=this.v813.formState;if(!s)return;
  const a=s[field]||[];
  const ix=a.map(String).indexOf(String(id));
  if(checked&&ix<0)a.push(id);
  if(!checked&&ix>=0)a.splice(ix,1);
  s[field]=a;
  if(field==='ourCalled')s.ourStarters=(s.ourStarters||[]).filter(x=>a.map(String).includes(String(x)));
  if(field==='rivalCalled')s.rivalStarters=(s.rivalStarters||[]).filter(x=>a.map(String).includes(String(x)));
  this.refreshRosterControls();
};

X.initFormState=function(e){
  if(e){
    const isLocal=e.home_away==='Local';
    const ownId=e.team_id||'', rivalId=e.opponent_team_id||'';
    const caps=e.captain_ids||[];
    this.v813.formState={
      id:e.id,round:e.round_label||'J1',category:e.category||'',ownTeamId:ownId,rivalTeamId:rivalId,
      gender:e.gender||'MASC',homeAway:e.home_away||'Local',status:e.status||'Programado',
      date:e.game_date||'',time:(e.game_time||'').slice(0,5),location:e.location||'',
      homeScore:e.home_score??'',awayScore:e.away_score??'',notes:e.notes||'',
      ourColor:isLocal?(e.home_color||''):(e.away_color||''),
      rivalColor:isLocal?(e.away_color||''):(e.home_color||''),
      ourCalled:(isLocal?e.home_called_ids:e.away_called_ids)||[],
      rivalCalled:(isLocal?e.away_called_ids:e.home_called_ids)||[],
      ourStarters:(isLocal?e.home_starter_ids:e.away_starter_ids)||[],
      rivalStarters:(isLocal?e.away_starter_ids:e.home_starter_ids)||[],
      ourCaptain:caps[0]||'',rivalCaptain:caps[1]||'',newLocation:''
    };
  }else{
    const cat=(document.getElementById('fsel-cat-enc')||{}).value||this.categoriesForSeason()[0]||'';
    const own=this.ownTeamsForCategory(cat)[0];
    const rival=this.rivalTeamsForCategory(cat,own&&own.id)[0];
    this.v813.formState={id:'',round:'J1',category:cat,ownTeamId:own?own.id:'',rivalTeamId:rival?rival.id:'',gender:'MASC',homeAway:'Local',status:'Programado',date:'',time:'',location:'',homeScore:'',awayScore:'',notes:'',ourColor:'',rivalColor:'',ourCalled:[],rivalCalled:[],ourStarters:[],rivalStarters:[],ourCaptain:'',rivalCaptain:'',newLocation:''};

X.applyTeamDefaults=function(){
  const s=this.v813.formState;if(!s)return;
  const own=this.teams[s.ownTeamId], rival=this.teams[s.rivalTeamId];
  if(!s.ourColor)s.ourColor=this.teamColors(own)[0];
  if(!s.rivalColor)s.rivalColor=this.teamColors(rival)[0];
  const local=s.homeAway==='Local'?own:rival;
  if(!s.location)s.location=this.teamVenue(local)||this.encLocations()[0]||'';
};

X.formChange=function(field,val){
  const s=this.v813.formState;if(!s)return;s[field]=val;
  if(field==='category'){
    const own=this.ownTeamsForCategory(val);
    if(!own.some(x=>String(x.id)===String(s.ownTeamId)))s.ownTeamId=own[0]?.id||'';
    const rivals=this.rivalTeamsForCategory(val,s.ownTeamId);
    if(!rivals.some(x=>String(x.id)===String(s.rivalTeamId)))s.rivalTeamId=rivals[0]?.id||'';
    s.ourCalled=[];s.ourStarters=[];s.ourCaptain='';s.rivalCalled=[];s.rivalStarters=[];s.rivalCaptain='';s.ourColor='';s.rivalColor='';s.location='';
    this.applyTeamDefaults();
  }
  if(field==='ownTeamId'){s.ourCalled=[];s.ourStarters=[];s.ourCaptain='';s.ourColor='';const rivals=this.rivalTeamsForCategory(s.category,s.ownTeamId);if(!rivals.some(x=>String(x.id)===String(s.rivalTeamId)))s.rivalTeamId=rivals[0]?.id||'';s.location='';this.applyTeamDefaults();}
  if(field==='rivalTeamId'){s.rivalCalled=[];s.rivalStarters=[];s.rivalCaptain='';s.rivalColor='';s.location='';this.applyTeamDefaults();}
  if(field==='homeAway'){s.location='';this.applyTeamDefaults();}
  this.render813Form();
};
